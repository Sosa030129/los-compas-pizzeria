   139→   139→function computeCartCount(state: AppState): number {
   140→   140→  return state.cart.reduce((n, c) => n + c.qty, 0);
   141→   141→}

   143→   143→function computeCartTotals(state: AppState): { subtotal: number; extras: number; delivery: number | null; total: number } {
   144→   144→  let subtotal = 0;
   145→   145→  let extras = 0;
   146→   146→  for (const item of state.cart) {
   147→   147→    subtotal += item.unitPrice * item.qty;
   148→   148→    extras += item.extrasTotal * item.qty;
   149→   149→  }
   150→   150→  const delivery =
   151→   151→    state.cart.length > 0 && state.cart.some((i) => !i.isCombo)
   152→   152→      ? null
   153→   153→      : 0;
   154→   154→  const base = subtotal + extras;
   155→   155→  const total = delivery === null ? base : base + delivery;
   156→   156→  return { subtotal, extras, delivery, total };
   157→   157→}

   751→   751→function isSameItem(a: CartItem, b: CartItem): boolean {
   752→   752→  if (a.productId !== b.productId) return false;
   753→   753→  if (a.size !== b.size) return false;
   754→   754→  if (a.borderCheese !== b.borderCheese) return false;
   755→   755→  if (a.isCombo !== b.isCombo) return false;
   756→   756→  const aIng = JSON.stringify(a.ingredients || []);
   757→   757→  const bIng = JSON.stringify(b.ingredients || []);
   758→   758→  if (aIng !== bIng) return false;
   759→   759→  if (a.notes !== b.notes) return false;
   760→   760→  return true;
   761→   761→}
