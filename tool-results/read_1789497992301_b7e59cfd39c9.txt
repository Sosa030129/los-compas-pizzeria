   114→// ===== Dashboard =====
   115→function DashboardTab({ onNavigate }: { onNavigate: (t: AdminTab) => void }) {
   116→  const orders = useStore((s) => s.orders);
   117→  const products = useStore((s) => s.products);
   118→  const ingredients = useStore((s) => s.ingredients);
   119→
   120→  const stats = {
   121→    nuevos: orders.filter((o) => o.state === 'recibido').length,
   122→    preparando: orders.filter((o) => o.state === 'preparando').length,
   123→    listos: orders.filter((o) => o.state === 'listo').length,
   124→    reparto: orders.filter((o) => o.state === 'camino').length,
   125→    entregados: orders.filter((o) => o.state === 'entregado').length,
   126→    cancelados: orders.filter((o) => o.state === 'cancelado').length,
   127→    ventas: orders.filter((o) => o.state === 'entregado').reduce((sum, o) => sum + o.total, 0),
   128→    agotados: products.filter((p) => !p.available).length,
   129→    ingredientesBaja: ingredients.filter((i) => !i.available).length,
   130→  };
   131→
   132→  const recent = orders.slice(0, 5);
   133→  const topProducts = products
   134→    .map((p) => {
   135→      const count = orders.flatMap((o) => o.items).filter((i) => i.productId === p.id).reduce((s, i) => s + i.qty, 0);
   136→      return { p, count };
   137→    })
   138→    .sort((a, b) => b.count - a.count)
   139→    .slice(0, 5);
   140→
   141→  return (
   142→    <div className="space-y-4">
   143→      {/* KPIs */}
   144→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   145→        <Stat label="Pedidos hoy" value={orders.length} emoji="📋" />
   146→        <Stat label="Ventas entr." value={formatCUP(stats.ventas)} emoji="💰" small />
   147→        <Stat label="Nuevos" value={stats.nuevos} emoji="🆕" highlight={stats.nuevos > 0} />
   148→        <Stat label="En reparto" value={stats.reparto} emoji="🛵" />
   149→      </div>
   150→
   151→      {/* Estados */}
   152→      <div className="cartoon-border bg-card rounded-2xl p-4">
   153→        <h3 className="font-cartoon text-sm mb-3">Estados actuales</h3>
   154→        <div className="grid grid-cols-3 gap-2 text-center">
   155→          {[
   156→            { label: 'Recibidos', value: stats.nuevos, color: '#8a7a5a' },
   157→            { label: 'Preparando', value: stats.preparando, color: '#d49050' },
   158→            { label: 'Listos', value: stats.listos, color: '#7ab860' },
   159→            { label: 'En camino', value: stats.reparto, color: '#5a9ab8' },
   160→            { label: 'Entregados', value: stats.entregados, color: '#7a1f2b' },
   161→            { label: 'Cancelados', value: stats.cancelados, color: '#7a1f1f' },
   162→          ].map((s) => (
   163→            <div key={s.label} className="bg-secondary/50 rounded-xl p-2">
   164→              <div className="text-lg font-bold" style={{ color: s.color }}>{s.value}</div>
   165→              <div className="text-[10px] text-muted-foreground">{s.label}</div>
   166→            </div>
   167→          ))}
   168→        </div>
   169→      </div>
   170→
   171→      {/* Gráficos avanzados */}
   172→      <StatsCharts />
   173→
   174→      {/* Alertas */}
   175→      {(stats.agotados > 0 || stats.ingredientesBaja > 0 || stats.nuevos > 0) && (
   176→        <div className="bg-yellow-500/10 border border-yellow-500/30 rounded-2xl p-4">
   177→          <h3 className="font-cartoon text-sm mb-2 flex items-center gap-1.5 text-yellow-400">
   178→            <AlertTriangle size={14} /> Alertas
   179→          </h3>
   180→          <ul className="text-xs text-yellow-200 space-y-1">
   181→            {stats.nuevos > 0 && (
   182→              <li>
   183→                📋 Tienes <strong>{stats.nuevos}</strong> pedido(s) nuevo(s) sin confirmar{' '}
   184→                <button onClick={() => onNavigate('orders')} className="underline text-yellow-300">ver →</button>
   185→              </li>
   186→            )}
   187→            {stats.agotados > 0 && <li>📦 {stats.agotados} producto(s) agotado(s)</li>}
   188→            {stats.ingredientesBaja > 0 && <li>🥬 {stats.ingredientesBaja} ingrediente(s) sin stock</li>}
   189→          </ul>
   190→        </div>
   191→      )}
   192→
   193→      {/* Pedidos recientes */}
   194→      <div className="cartoon-border bg-card rounded-2xl p-4">
   195→        <div className="flex items-center justify-between mb-3">
   196→          <h3 className="font-cartoon text-sm">Pedidos recientes</h3>
   197→          <button onClick={() => onNavigate('orders')} className="text-xs text-primary font-bold hover:underline">
   198→            Ver todo →
   199→          </button>
   200→        </div>
   201→        {recent.length === 0 ? (
   202→          <p className="text-xs text-muted-foreground text-center py-4">Sin pedidos todavía</p>
   203→        ) : (
   204→          <div className="space-y-2">
   205→            {recent.map((o) => {
   206→              const st = getStateInfo(o.state);
   207→              return (
   208→                <button
   209→                  key={o.id}
   210→                  onClick={() => onNavigate('orders')}
   211→                  className="w-full flex items-center gap-3 text-left hover:bg-accent/20 rounded-xl p-2"
   212→                >
   213→                  <span className="font-bold text-xs text-primary w-16">{o.code}</span>
   214→                  <div className="flex-1 min-w-0">
   215→                    <p className="text-xs truncate">{o.customerName}</p>
   216→                    <p className="text-[10px] text-muted-foreground">{formatDateTime(o.createdAt)}</p>
   217→                  </div>
   218→                  <span className="text-[10px] font-bold px-2 py-0.5 rounded-full"
   219→                    style={{ backgroundColor: `${st.color}33`, color: st.color }}>
   220→                    {st.emoji} {st.label}
   221→                  </span>
   222→                  <span className="text-xs font-bold">{formatCUP(o.total)}</span>
   223→                </button>
   224→              );
   225→            })}
   226→          </div>
   227→        )}
   228→      </div>
   229→
   230→      {/* Top productos */}
   231→      <div className="cartoon-border bg-card rounded-2xl p-4">
   232→        <h3 className="font-cartoon text-sm mb-3">Productos más vendidos</h3>
   233→        {topProducts.every((p) => p.count === 0) ? (
   234→          <p className="text-xs text-muted-foreground text-center py-4">Sin ventas registradas</p>
   235→        ) : (
   236→          <div className="space-y-1.5">
   237→            {topProducts.filter((p) => p.count > 0).map(({ p, count }) => (
   238→              <div key={p.id} className="flex items-center gap-2 text-sm">
   239→                <span className="text-lg">{p.emoji}</span>
   240→                <span className="flex-1 truncate">{p.name}</span>
   241→                <span className="text-xs font-bold text-primary">{count}×</span>
   242→              </div>
   243→            ))}
   244→          </div>
   245→        )}
   246→      </div>
   247→    </div>
   248→  );
   249→}

   252→  label: string; value: string | number; emoji: string; highlight?: boolean; small?: boolean;
   253→}) {

   259→    </div>
   260→  );

   264→function OrdersTab() {
   265→  const orders = useStore((s) => s.orders);
   266→  const setOrderState = useStore((s) => s.setOrderState);
   267→  const setOrderDelivery = useStore((s) => s.setOrderDelivery);
   268→  const assignDelivery = useStore((s) => s.assignDelivery);
   269→  const employees = useStore((s) => s.employees);
   270→  const sizes = useStore((s) => s.sizes);
   271→
   272→  const [filter, setFilter] = useState<string>('all');
   273→  const [editDelivery, setEditDelivery] = useState<string | null>(null);
   274→  const [deliveryInput, setDeliveryInput] = useState('');
   275→
   276→  const filtered = filter === 'all' ? orders : orders.filter((o) => o.state === filter);
   277→  const repartidores = employees.filter((e) => e.role === 'repartidor' && e.active);
   278→
   279→  return (
   280→    <div className="space-y-3">
   281→      <div className="flex gap-1 overflow-x-auto -mx-4 px-4">
   282→        {['all', 'recibido', 'confirmado', 'preparando', 'listo', 'camino', 'entregado', 'cancelado'].map((s) => (
   283→          <button
   284→            key={s}
   285→            onClick={() => setFilter(s)}
   286→            className={`shrink-0 px-3 py-1.5 rounded-full text-xs font-bold ${
   287→              filter === s ? 'bg-primary text-primary-foreground' : 'bg-secondary text-secondary-foreground'
   288→            }`}
   289→          >
   290→            {s === 'all' ? 'Todos' : getStateInfo(s).label}
   291→          </button>
   292→        ))}
   293→      </div>
   294→
   295→      {filtered.length === 0 ? (
   296→        <div className="text-center py-12 text-muted-foreground">
   297→          <p className="text-sm">No hay pedidos en este estado</p>
   298→        </div>
   299→      ) : (
   300→        filtered.map((o) => {
   301→          const st = getStateInfo(o.state);
   302→          return (
   303→            <div key={o.id} className="cartoon-border bg-card rounded-2xl p-4">
   304→              <div className="flex items-start justify-between mb-2">
   305→                <div>
   306→                  <div className="flex items-center gap-2">
   307→                    <span className="font-cartoon text-sm text-primary">{o.code}</span>
   308→                    <span className="text-[10px] font-bold px-2 py-0.5 rounded-full"
   309→                      style={{ backgroundColor: `${st.color}33`, color: st.color }}>
   310→                      {st.emoji} {st.label}
   311→                    </span>
   312→                  </div>
   313→                  <p className="text-xs text-muted-foreground mt-0.5">
   314→                    {o.customerName} · {o.customerPhone}
   315→                  </p>
   316→                  {o.deliveryMode === 'domicilio' && (
   317→                    <p className="text-[11px] text-muted-foreground mt-0.5">📍 {o.customerAddress}</p>
   318→                  )}
   319→                  <p className="text-[11px] text-muted-foreground">🕐 {o.scheduledTime}</p>
   320→                </div>
   321→                <div className="text-right">
   322→                  <p className="font-cartoon text-base text-primary">{formatCUP(o.total)}</p>
   323→                  <p className="text-[10px] text-muted-foreground">
   324→                    {o.paymentMethod === 'efectivo' ? '💵 Efectivo' : '💳 Transferencia'}
   325→                  </p>
   326→                </div>
   327→              </div>
   328→
   329→              <div className="space-y-1 mb-3 bg-secondary/40 rounded-xl p-2">
   330→                {o.items.map((item, i) => {
   331→                  const size = item.size ? sizes.find((s) => s.id === item.size) : null;
   332→                  return (
   333→                    <div key={i} className="text-xs flex items-start gap-2">
   334→                      <span>{item.emoji}</span>
   335→                      <span className="flex-1">
   336→                        {item.qty}× {item.name}
   337→                        {size && ` · ${size.label}`}
   338→                        {item.borderCheese && ' · Borde queso'}
   339→                      </span>
   340→                      <span className="text-muted-foreground">
   341→                        {formatCUP((item.unitPrice + item.extrasTotal) * item.qty)}
   342→                      </span>
   343→                    </div>
   344→                  );
   345→                })}
   346→              </div>
   347→
   348→              {/* Domicilio */}
   349→              {o.deliveryMode === 'domicilio' && (
   350→                <div className="bg-secondary/40 rounded-xl p-2 mb-2 flex items-center justify-between">
   351→                  <span className="text-xs text-muted-foreground">Domicilio</span>
   352→                  {editDelivery === o.id ? (
   353→                    <div className="flex items-center gap-1">
   354→                      <input
   355→                        type="number"
   356→                        value={deliveryInput}
   357→                        onChange={(e) => setDeliveryInput(e.target.value)}
   358→                        className="w-20 bg-background border border-border rounded px-2 py-1 text-xs"
   359→                      />
   360→                      <button
   361→                        onClick={() => {
   362→                          const v = parseInt(deliveryInput);
   363→                          if (!isNaN(v) && v >= 0) {
   364→                            setOrderDelivery(o.id, v);
   365→                            toast.success(`Domicilio actualizado: ${formatCUP(v)}`);
   366→                          }
   367→                          setEditDelivery(null);
   368→                        }}
   369→                        className="bg-primary text-primary-foreground px-2 py-1 rounded text-xs"
   370→                      >
   371→                        ✓
   372→                      </button>
   373→                      <button
   374→                        onClick={() => setEditDelivery(null)}
   375→                        className="bg-secondary px-2 py-1 rounded text-xs"
   376→                      >
   377→                        ✕
   378→                      </button>
   379→                    </div>
   380→                  ) : (
   381→                    <button
   382→                      onClick={() => {
   383→                        setDeliveryInput(String(o.delivery ?? 0));
   384→                        setEditDelivery(o.id);
   385→                      }}
   386→                      className="text-xs font-bold text-primary hover:underline"
   387→                    >
   388→                      {o.delivery === null ? 'Pendiente — Click para confirmar' : formatCUP(o.delivery)}
   389→                    </button>
   390→                  )}
   391→                </div>
   392→              )}
   393→
   394→              {/* Acciones */}
   395→              <div className="flex flex-wrap gap-1.5">
   396→                {o.state === 'recibido' && (
   397→                  <button
   398→                    onClick={() => {
   399→                      setOrderState(o.id, 'confirmado');
   400→                      toast.success('Pedido confirmado');
   401→                    }}
   402→                    className="bg-primary text-primary-foreground px-3 py-1.5 rounded-full text-xs font-bold"
   403→                  >
   404→                    ✓ Confirmar
   405→                  </button>
   406→                )}
   407→                {o.state === 'confirmado' && (
   408→                  <button
   409→                    onClick={() => {
   410→                      setOrderState(o.id, 'preparando');
   411→                      toast.success('Marcado como preparando');
   412→                    }}
   413→                    className="bg-accent text-accent-foreground px-3 py-1.5 rounded-full text-xs font-bold"
   414→                  >
   415→                    👨‍🍳 Iniciar preparación
   416→                  </button>
   417→                )}
   418→                {o.state === 'preparando' && (
   419→                  <button
   420→                    onClick={() => {
   421→                      setOrderState(o.id, 'listo');
   422→                      toast.success('Pedido listo');
   423→                    }}
   424→                    className="bg-green-700 text-white px-3 py-1.5 rounded-full text-xs font-bold"
   425→                  >
   426→                    📦 Listo
   427→                  </button>
   428→                )}
   429→                {o.state === 'listo' && o.deliveryMode === 'domicilio' && (
   430→                  <select
   431→                    onChange={(e) => {
   432→                      if (e.target.value) {
   433→                        assignDelivery(o.id, e.target.value);
   434→                        toast.success('Repartidor asignado');
   435→                      }
   436→                    }}
   437→                    defaultValue=""
   438→                    className="bg-secondary text-secondary-foreground px-3 py-1.5 rounded-full text-xs font-bold border border-border"
   439→                  >
   440→                    <option value="">Asignar repartidor...</option>
   441→                    {repartidores.map((r) => (
   442→                      <option key={r.id} value={r.id}>{r.name}</option>
   443→                    ))}
   444→                  </select>
   445→                )}
   446→                {o.state === 'listo' && o.deliveryMode === 'recogida' && (
   447→                  <button
   448→                    onClick={() => {
   449→                      setOrderState(o.id, 'entregado');
   450→                      toast.success('Pedido entregado');
   451→                    }}
   452→                    className="bg-primary text-primary-foreground px-3 py-1.5 rounded-full text-xs font-bold"
   453→                  >
   454→                    🎉 Entregar
   455→                  </button>
   456→                )}
   457→                {o.state !== 'entregado' && o.state !== 'cancelado' && (
   458→                  <button
   459→                    onClick={() => {
   460→                      setOrderState(o.id, 'cancelado');
   461→                      toast('Pedido cancelado');
   462→                    }}
   463→                    className="bg-destructive/20 text-destructive px-3 py-1.5 rounded-full text-xs font-bold"
   464→                  >
   465→                    ✕ Cancelar
   466→                  </button>
   467→                )}
   468→              </div>
   469→            </div>
   470→          );
   471→        })
   472→      )}
   473→    </div>
   474→  );
   475→}

  1713→function ComboForm({
  1714→  initial, products, onClose, onSave,
  1715→}: {
  1716→  initial: Product | null;
  1717→  products: Product[];
  1718→  onClose: () => void;
  1719→  onSave: (p: Product) => void;

  1723→  const [emoji, setEmoji] = useState(initial?.emoji || '🎉');
  1724→  const [price, setPrice] = useState(initial?.price || 0);
  1725→  // Cantidad de cada producto en el combo: { productId: qty }
  1726→  const [items, setItems] = useState<Record<string, number>>(() => {
  1727→    const init: Record<string, number> = {};
  1728→    if (initial?.comboItems) {
  1729→      for (const pid of initial.comboItems) {
  1730→        init[pid] = (init[pid] || 0) + 1;
  1731→      }
  1732→    }
  1733→    return init;
  1734→  });
  1735→
  1736→  // Precio individual sumado de los productos seleccionados
  1737→  const precioIndividual = Object.entries(items).reduce((sum, [pid, qty]) => {
  1738→    const p = products.find((x) => x.id === pid);
  1739→    return sum + (p ? p.price * qty : 0);
  1740→  }, 0);
  1741→  const ahorro = precioIndividual - price;
  1742→  const ahorroPct = precioIndividual > 0 ? Math.round((ahorro / precioIndividual) * 100) : 0;
  1743→
  1744→  const toggleProduct = (pid: string) => {
  1745→    setItems((prev) => {
  1746→      const next = { ...prev };
  1747→      if (next[pid]) {
  1748→        next[pid] = next[pid] + 1;
  1749→      } else {
  1750→        next[pid] = 1;
  1751→      }
  1752→      return next;
  1753→    });
  1754→  };
  1755→
  1756→  const decrement = (pid: string) => {
  1757→    setItems((prev) => {
  1758→      const next = { ...prev };
  1759→      if (next[pid]) {
  1760→        next[pid] = next[pid] - 1;
  1761→        if (next[pid] <= 0) delete next[pid];
  1762→      }
  1763→      return next;
  1764→    });
  1765→  };
  1766→
  1767→  const totalSelected = Object.values(items).reduce((s, q) => s + q, 0);
  1768→
  1769→  return (
  1770→    <div className="fixed inset-0 z-50 bg-black/60 backdrop-blur-sm flex items-end sm:items-center justify-center p-4">
  1771→      <div className="bg-card rounded-3xl p-5 w-full max-w-md border-2 border-border max-h-[90vh] overflow-y-auto">
  1772→        <h3 className="font-cartoon text-base mb-3">{initial ? 'Editar combo' : 'Nuevo combo'}</h3>
  1773→        <div className="space-y-3">
  1774→          <div className="grid grid-cols-[auto_1fr] gap-2">
  1775→            <FormRow label="Emoji">
  1776→              <input value={emoji} onChange={(e) => setEmoji(e.target.value)} className="bg-background border border-border rounded-xl px-3 py-2 w-full text-sm text-center" />
  1777→            </FormRow>
  1778→            <FormRow label="Nombre del combo">
  1779→              <input value={name} onChange={(e) => setName(e.target.value)} placeholder="Combo Familiar" className="bg-background border border-border rounded-xl px-3 py-2 w-full text-sm" />
  1780→            </FormRow>
  1781→          </div>
  1782→          <FormRow label="Descripción">
  1783→            <textarea value={description} onChange={(e) => setDescription(e.target.value)} placeholder="Pizza + 2 bebidas + postre" className="bg-background border border-border rounded-xl px-3 py-2 w-full text-sm min-h-[60px]" />
  1784→          </FormRow>
  1785→
  1786→          {/* Selector de productos */}
  1787→          <div>
  1788→            <p className="text-[11px] font-bold text-muted-foreground mb-2">
  1789→              Productos incluidos ({totalSelected} items):
  1790→            </p>
  1791→            <div className="max-h-64 overflow-y-auto space-y-1.5 bg-background/40 rounded-xl p-2">
