   138→// ===== Helpers libres de efectos (computan sobre el estado, no lo mutan) =====
   139→function computeCartCount(state: AppState): number {
   140→  return state.cart.reduce((n, c) => n + c.qty, 0);
   141→}
   142→
   143→function computeCartTotals(state: AppState): { subtotal: number; extras: number; delivery: number | null; total: number } {
   144→  let subtotal = 0;
   145→  let extras = 0;
   146→  for (const item of state.cart) {
   147→    subtotal += item.unitPrice * item.qty;
   148→    extras += item.extrasTotal * item.qty;
   149→  }
   150→  const delivery =
   151→    state.cart.length > 0 && state.cart.some((i) => !i.isCombo)
   152→      ? null
   153→      : 0;
   154→  const base = subtotal + extras;
   155→  const total = delivery === null ? base : base + delivery;
   156→  return { subtotal, extras, delivery, total };
   157→}

   159→// Cap de logs para evitar localStorage overflow (max 200)

   551→          });

   553→          if (!result.ok) {
   554→            registerFailedLogin(username);
   555→            return { ok: false, error: result.error || 'Credenciales inválidas' };
   556→          }
   557→          // Login exitoso: el servidor ya seteó la cookie httpOnly
   558→          resetLoginAttempts(username);
   559→          const emp = result.employee as Employee;

   568→                date: Date.now(),
   569→              },
   570→              ...s.logs,
   571→            ]),
   572→          }));
   573→          return { ok: true, error: null };
   574→        } catch (e) {
   575→          // Fallback: intentar login local (modo demo offline)
   576→          const emp = get().employees.find(
   577→            (e) => e.username.toLowerCase() === username.toLowerCase() && e.active
   578→          );
   579→          if (!emp) {
   580→            registerFailedLogin(username);
   581→            return { ok: false, error: 'Usuario o contraseña incorrectos' };
   582→          }
   583→          const passwordOk = await verifyPassword(password, emp.password);
   584→          if (!passwordOk) {
   585→            registerFailedLogin(username);
   586→            return { ok: false, error: 'Usuario o contraseña incorrectos' };
   587→          }
   588→          resetLoginAttempts(username);
   589→          set((s) => ({

   751→function isSameItem(a: CartItem, b: CartItem): boolean {
   752→  if (a.productId !== b.productId) return false;
   753→  if (a.size !== b.size) return false;
   754→  if (a.borderCheese !== b.borderCheese) return false;
   755→  if (a.isCombo !== b.isCombo) return false;
   756→  const aIng = JSON.stringify(a.ingredients || []);
   757→  const bIng = JSON.stringify(b.ingredients || []);
   758→  if (aIng !== bIng) return false;
   759→  if (a.notes !== b.notes) return false;
   760→  return true;
   761→}

   767→// ===== Funciones de hidratación desde el backend =====
   768→// Carga el catálogo (productos, categorías, ingredientes, tamaños, promociones, config, whatsapp)
   769→// desde /api/catalog y actualiza el store. Útil al montar la app.
   770→export async function hydrateFromServer(): Promise<boolean> {
   771→  try {
   772→    const res = await fetch('/api/catalog');
   773→    if (!res.ok) return false;
   774→    const result = await res.json();
   775→    if (!result.ok || !result.catalog) return false;
   776→
   777→    const c = result.catalog;
   778→    const patch: any = {};
   779→    if (Array.isArray(c.categories)) patch.categories = c.categories;
   780→    if (Array.isArray(c.products)) patch.products = c.products;
   781→    if (Array.isArray(c.ingredients)) patch.ingredients = c.ingredients;
   782→    if (Array.isArray(c.sizes)) patch.sizes = c.sizes;
   783→    if (Array.isArray(c.promotions)) patch.promotions = c.promotions;
   784→    if (Array.isArray(c.whatsappNumbers)) patch.whatsapp = c.whatsappNumbers;
   785→    if (c.config) patch.config = c.config;
   786→
   787→    if (Array.isArray(c.products)) {
   788→      patch.combos = c.products.filter((p: any) => p.isCombo);
   789→    }
   790→
   791→    useStore.setState(patch);
   792→    return true;
   793→  } catch (e) {
   794→    return false;
   795→  }
   796→}

   800→export async function refreshOrders(): Promise<boolean> {
   801→  try {
   802→    const res = await fetch('/api/orders');
   803→    if (!res.ok) return false;
   804→    const result = await res.json();
   805→    if (!result.ok || !Array.isArray(result.orders)) return false;
   806→    // Parsear items JSON string en cada pedido
   807→    const orders = result.orders.map((o: any) => ({
   808→      ...o,
   809→      items: typeof o.items === 'string' ? JSON.parse(o.items) : o.items,
   810→      createdAt: typeof o.createdAt === 'string' ? new Date(o.createdAt).getTime() : o.createdAt,
   811→      confirmedAt: o.confirmedAt ? (typeof o.confirmedAt === 'string' ? new Date(o.confirmedAt).getTime() : o.confirmedAt) : undefined,
   812→      deliveredAt: o.deliveredAt ? (typeof o.deliveredAt === 'string' ? new Date(o.deliveredAt).getTime() : o.deliveredAt) : undefined,
   813→      validFrom: undefined, validTo: undefined,
   814→    }));
   815→    useStore.setState({ orders });
   816→    return true;
   817→  } catch (e) {
   818→    return false;
   819→  }
